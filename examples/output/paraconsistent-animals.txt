?- wings_safely(batsy, undecided).
true.

?- why.
wheye-proof 1
% Rule 4: wings(X, false) :- mammal(X).
% Rule 6: wings(X, true) :- bat(X).
% Rule 13: wing_status(X, both) :- wings(X, true), wings(X, false).
% Rule 25: wings_safely(X, undecided) :- wing_status(X, both).
step 1 source 5 : mammal(batsy).
step 2 source 6 : bat(batsy).
step 3 rule 4 [1] : wings(batsy, false).
step 4 rule 6 [2] : wings(batsy, true).
step 5 rule 13 [4,3] : wing_status(batsy, both).
step 6 rule 25 [5] : wings_safely(batsy, undecided).
answer 3 [6]
