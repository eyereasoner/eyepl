?- append([a, b], [c, d], Joined).
Joined = [a, b, c, d].

?- why.
wheye-proof 1
% Rule 1: append([], Ys, Ys) :- .
% Rule 2: append([X | Xs], Ys, [X | Zs]) :- append(Xs, Ys, Zs).
step 1 rule 1 [] : append([], [c, d], [c, d]).
step 2 rule 2 [1] : append([b], [c, d], [b, c, d]).
step 3 rule 2 [2] : append([a, b], [c, d], [a, b, c, d]).
answer 1 [3]

?- squares([1, 2, 3, 4], Squared).
Squared = [1, 4, 9, 16].

?- why.
wheye-proof 1
% Rule 3: squares([X | Xs], [Y | Ys]) :- Y is '*'(X, X), squares(Xs, Ys).
step 1 source 1 : squares([], []).
step 2 builtin : is(16, '*'(4, 4)).
step 3 builtin : is(9, '*'(3, 3)).
step 4 builtin : is(4, '*'(2, 2)).
step 5 builtin : is(1, '*'(1, 1)).
step 6 rule 3 [2,1] : squares([4], [16]).
step 7 rule 3 [3,6] : squares([3, 4], [9, 16]).
step 8 rule 3 [4,7] : squares([2, 3, 4], [4, 9, 16]).
step 9 rule 3 [5,8] : squares([1, 2, 3, 4], [1, 4, 9, 16]).
answer 2 [9]

?- sum([1, 2, 3, 4], Total).
Total = 10.

?- why.
wheye-proof 1
% Rule 4: sum([X | Xs], Total) :- sum(Xs, Rest), Total is '+'(X, Rest).
step 1 source 2 : sum([], 0).
step 2 builtin : is(4, '+'(4, 0)).
step 3 builtin : is(7, '+'(3, 4)).
step 4 builtin : is(9, '+'(2, 7)).
step 5 builtin : is(10, '+'(1, 9)).
step 6 rule 4 [1,2] : sum([4], 4).
step 7 rule 4 [6,3] : sum([3, 4], 7).
step 8 rule 4 [7,4] : sum([2, 3, 4], 9).
step 9 rule 4 [8,5] : sum([1, 2, 3, 4], 10).
answer 3 [9]
